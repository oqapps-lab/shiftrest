/**
 * Pure calendar-grid builders. Extracted from app/(tabs)/schedule.tsx
 * to make them unit-testable (the previous in-screen version had a
 * silent negative-modulo bug for future months — caught by audit in
 * Stage 6.7 — so locking in tests prevents regression).
 */
 
export type ShiftKind = 'day' | 'night' | 'off' | 'past' | 'empty';
 
export interface Cell {
  label: number | '';
  kind: ShiftKind;
  iso?: string;
}
 
/** YYYY-MM-DD using local timezone (not UTC). */
export function localIso(d: Date): string {
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${y}-${m}-${day}`;
}
 
/**
 * 6-week (42-cell) Monday-first grid for a given month, painted from
 * a user's real shift data. Empty cells fill the leading offset; days
 * before today render as 'past'; days without a real shift render as 'off'.
 */
export function buildMonthGrid(
  year: number,
  month: number,
  shiftByIso: Map<string, 'day' | 'night'>,
  today: Date = new Date(),
): Cell[] {
  const firstOfMonth = new Date(year, month, 1);
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  // Monday-first offset: getDay() returns 0=Sun..6=Sat → shift to 0=Mon..6=Sun
  const offset = (firstOfMonth.getDay() + 6) % 7;
  const todayIso = localIso(today);
 
  const cells: Cell[] = [];
  for (let i = 0; i < offset; i++) cells.push({ label: '', kind: 'empty' });
 
  for (let d = 1; d <= daysInMonth; d++) {
    const date = new Date(year, month, d);
    const iso = localIso(date);
    const realKind = shiftByIso.get(iso);
    let kind: ShiftKind;
    if (realKind) {
      kind = realKind;
    } else if (iso < todayIso) {
      kind = 'past';
    } else {
      kind = 'off';
    }
    cells.push({ label: d, kind, iso });
  }
 
  while (cells.length % 7 !== 0) cells.push({ label: '', kind: 'empty' });
  return cells;
}
 
/**
 * Demo fallback grid for unauthenticated users — paints a 12-day rotation
 * cycle so the calendar tells a story before signup.
 *
 * Regression note: the negative-modulo guard `((x%n)+n)%n` is critical.
 * Without it, JS's modulo returns negative for `d < today.getDate()` on
 * future months, indexing past array start → undefined → silently rendered
 * every future day as 'off'. Tested via `__tests__/schedule.test.ts`.
 */
export function buildMockGrid(year: number, month: number, today: Date = new Date()): Cell[] {
  const firstOfMonth = new Date(year, month, 1);
  const daysInMonth = new Date(year, month + 1, 0).getDate();
  const offset = (firstOfMonth.getDay() + 6) % 7;
  const todayIso = localIso(today);
  const cycle: ('day' | 'night' | 'off')[] = [
    'day', 'day', 'day', 'off', 'off', 'night', 'night',
    'night', 'off', 'off', 'day', 'day',
  ];
  const cells: Cell[] = [];
  for (let i = 0; i < offset; i++) cells.push({ label: '', kind: 'empty' });
  const msPerDay = 86400000;
  const todayMidnight = new Date(today.getFullYear(), today.getMonth(), today.getDate()).getTime();
  for (let d = 1; d <= daysInMonth; d++) {
    const iso = localIso(new Date(year, month, d));
    if (iso < todayIso) {
      cells.push({ label: d, kind: 'past', iso });
      continue;
    }
    const daysFromToday = Math.round((new Date(year, month, d).getTime() - todayMidnight) / msPerDay);
    const idx = ((daysFromToday % cycle.length) + cycle.length) % cycle.length;
    cells.push({ label: d, kind: cycle[idx], iso });
  }
  while (cells.length % 7 !== 0) cells.push({ label: '', kind: 'empty' });
  return cells;
}
